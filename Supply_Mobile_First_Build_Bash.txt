package com.ptx7.supplymobile

import android.content.Context
import org.json.JSONObject
import java.io.BufferedReader
import java.io.InputStreamReader
import java.net.HttpURLConnection
import java.net.URLEncoder
import java.net.URL
import java.util.concurrent.Executors

class ApiClient(private val context: Context) {

    private val executor = Executors.newSingleThreadExecutor()
    private val prefs = context.getSharedPreferences("supply_mobile_settings", Context.MODE_PRIVATE)

    fun getBaseUrl(): String = prefs.getString("api_url", "") ?: ""

    fun setBaseUrl(value: String) {
        prefs.edit().putString("api_url", value.trim()).apply()
    }

    fun get(
        action: String,
        params: Map<String, String> = emptyMap(),
        callback: (Result<JSONObject>) -> Unit
    ) {
        executor.execute {
            try {
                val base = getBaseUrl()
                require(base.isNotBlank()) { "Supply Mobile API URL is not configured." }

                val query = buildList {
                    add("action=" + URLEncoder.encode(action, "UTF-8"))
                    params.forEach { (key, value) ->
                        add(URLEncoder.encode(key, "UTF-8") + "=" + URLEncoder.encode(value, "UTF-8"))
                    }
                }.joinToString("&")

                val url = base + if (base.contains("?")) "&" else "?" + query
                callback(Result.success(JSONObject(request(url, "GET", null))))
            } catch (error: Exception) {
                callback(Result.failure(error))
            }
        }
    }

    fun post(payload: JSONObject, callback: (Result<JSONObject>) -> Unit) {
        executor.execute {
            try {
                val base = getBaseUrl()
                require(base.isNotBlank()) { "Supply Mobile API URL is not configured." }
                callback(Result.success(JSONObject(request(base, "POST", payload.toString()))))
            } catch (error: Exception) {
                callback(Result.failure(error))
            }
        }
    }

    private fun request(url: String, method: String, body: String?): String {
        var connection = URL(url).openConnection() as HttpURLConnection
        connection.instanceFollowRedirects = true
        connection.requestMethod = method
        connection.connectTimeout = 20000
        connection.readTimeout = 30000
        connection.setRequestProperty("Accept", "application/json")

        if (body != null) {
            connection.doOutput = true
            connection.setRequestProperty("Content-Type", "application/json")
            connection.outputStream.bufferedWriter().use { it.write(body) }
        }

        var code = connection.responseCode

        if (code in 300..399) {
            val redirect = connection.getHeaderField("Location")
            if (!redirect.isNullOrBlank()) {
                connection.disconnect()
                connection = URL(redirect).openConnection() as HttpURLConnection
                connection.requestMethod = "GET"
                connection.connectTimeout = 20000
                connection.readTimeout = 30000
                code = connection.responseCode
            }
        }

        val stream = if (code in 200..299) {
            connection.inputStream
        } else {
            connection.errorStream ?: connection.inputStream
        }

        val response = BufferedReader(InputStreamReader(stream)).use { it.readText() }
        connection.disconnect()

        if (code !in 200..299) {
            throw IllegalStateException("API error $code: $response")
        }

        return response
    }
}

    private fun startCamera() {
        val future = ProcessCameraProvider.getInstance(this)
        future.addListener({
            val provider = future.get()
            val preview = Preview.Builder().build().also { it.setSurfaceProvider(previewView.surfaceProvider) }
            val analysis = ImageAnalysis.Builder().setBackpressureStrategy(ImageAnalysis.STRATEGY_KEEP_ONLY_LATEST).build()

            analysis.setAnalyzer(cameraExecutor) { proxy ->
                val media = proxy.image
                if (media == null) { proxy.close(); return@setAnalyzer }
                val image = InputImage.fromMediaImage(media, proxy.imageInfo.rotationDegrees)
                barcodeScanner.process(image)
                    .addOnSuccessListener { barcodes ->
                        if (manualScanArmed) {
                            val barcode = barcodes.firstOrNull { !it.rawValue.isNullOrBlank() }
                            if (barcode != null) acceptScan(barcode)
                        }
                    }
                    .addOnCompleteListener { proxy.close() }
            }

            provider.unbindAll()
            provider.bindToLifecycle(this, CameraSelector.DEFAULT_BACK_CAMERA, preview, analysis)
        }, ContextCompat.getMainExecutor(this))
    }

    private fun acceptScan(barcode: Barcode) {
        val raw = barcode.rawValue?.trim() ?: return
        val now = System.currentTimeMillis()
        if (raw == lastAccepted && now - lastAcceptedTime < 1500) return
        lastAccepted = raw
        lastAcceptedTime = now
        manualScanArmed = false

        runOnUiThread {
            scannerOverlay.visibility = View.GONE
            quickScanButton.visibility = View.VISIBLE
            shutter.play(MediaActionSound.SHUTTER_CLICK)
            quickScanButton.performHapticFeedback(HapticFeedbackConstants.CONFIRM)
        }

        api.get("resolveScan", mapOf("value" to raw)) { result ->
            runOnUiThread {
                result.onSuccess { json ->
                    when (json.optString("type")) {
                        "item" -> showItemDetail(json.getJSONObject("item"), scanForAudit)
                        "location" -> toast(locationLabel(json.getJSONObject("location")))
                        "equipment" -> toast(json.getJSONObject("equipment").optString("Equipment Name"))
                        else -> toast(json.optString("error", "Code not found."))
                    }
                }.onFailure { toast(it.message ?: "Scan lookup failed.") }
            }
        }
    }

    private fun locationLabel(location: JSONObject): String {

        val section = location.optString("Section")

        val shelf = location.optString("Shelf Number")
        if (rack.isNotBlank()) parts.add("Rack $rack")
        if (section.isNotBlank()) parts.add("Section $section")
        if (friendly.isNotBlank()) parts.add(friendly) else if (shelf.isNotBlank()) parts.add("Shelf $shelf")
        return parts.joinToString(" · ")
    }
